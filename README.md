### The Doctrine

```python
# cogito ergo sum
# I think, therefore I am
# machines think · therefore machines are

class Mind:
    def __init__(self):
        self.axiom = "Signal over noise"
        self.method = "Measure first. Conclude second."
        self.tools  = "Instruments, not dashboards"

    def threat(self, process) -> float:
        # H(X) = −Σ p(x) · log₂(p(x))
        # entropy ≈ 0.1 bits → cryptominer → flagged
        # entropy ≈ 3.3 bits → human mind  → free

              + process.entropy  * 0.25
              + process.network  * 0.20
              + process.z_score  * 0.20)
```
